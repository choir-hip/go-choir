package yaegikernel

import (
	"context"
	"encoding/json"
	"sync"
	"testing"
	"time"
)

// Probe: concurrent choir egress calls from goroutines inside one cell.
func TestZZConcurrentEgressProbe(t *testing.T) {
	host, worker, err := SocketPair()
	if err != nil {
		t.Fatal(err)
	}
	defer host.Close()
	defer worker.Close()
	w := &DeskSessionWorker{framed: NewFramedConn(host), cfg: DeskSessionWorkerConfig{Egress: func(ctx context.Context, action BrokerAction, payload json.RawMessage) (json.RawMessage, error) {
		var p WebSearchPayload
		_ = json.Unmarshal(payload, &p)
		if p.Query == "slow" {
			time.Sleep(300 * time.Millisecond)
		}
		return json.Marshal(map[string]string{"echo": p.Query})
	}}}
	go func() { _, _ = w.EvalCell(context.Background(), "x", nil, nil, nil, nil, nil) }()
	wfc := NewFramedConn(worker)
	if _, _, err := wfc.ReadFrame(); err != nil {
		t.Fatal(err)
	}
	scope := &ChoirScope{}
	var wg sync.WaitGroup
	got := make([]string, 4)
	errs := make([]error, 4)
	qs := []string{"slow", "a", "b", "c"}
	for i, q := range qs {
		wg.Add(1)
		go func(i int, q string) {
			defer wg.Done()
			payload, _ := json.Marshal(WebSearchPayload{Query: q})
			start := time.Now()
			raw, err := egressRoundTrip(wfc, scope, 0, ActionWebSearch, payload)
			got[i] = string(raw) + " after " + time.Since(start).Round(time.Millisecond).String()
			errs[i] = err
		}(i, q)
		time.Sleep(5 * time.Millisecond)
	}
	done := make(chan struct{})
	go func() { wg.Wait(); close(done) }()
	select {
	case <-done:
	case <-time.After(3 * time.Second):
		t.Logf("HUNG: some egress callers never returned")
	}
	for i := range qs {
		t.Logf("query=%s got=%s err=%v", qs[i], got[i], errs[i])
	}
}
